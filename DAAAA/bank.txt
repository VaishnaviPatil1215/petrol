pragma solidity ^0.8.0;

contract Bank {

   uint256 balance=0;
   address public accOwner;

   constructor(){
       accOwner=msg.sender;
   }

    function deposit() public payable {

        require(accOwner==msg.sender,"You are not an account owner");
        require(msg.value > 0,"Amount should be greater than zero!");

        balance+=msg.value;
    }

    function withdraw() public payable{

        require(accOwner==msg.sender,"You are not an account owner");
        require(msg.value>0,"Withdrawal money should be greater than zero");

        balance-=msg.value;
    }

    function showBalance()public view returns(uint256){
        require(accOwner==msg.sender,"You are not an account owner");
        return balance;
    }
}

7. Important corrections to make this a proper Bank contract
Your current code demonstrates the basic logic, but there are three important issues:
1. The withdrawal function should transfer Ether to the owner instead of asking the owner to send more Ether.
2. We should check whether sufficient balance is available before withdrawal.
3. The contract's actual Ether balance and the internal balance variable should remain consistent.
A better withdrawal function would look like this:
function withdraw(uint256 amount) public {

    require(accOwner == msg.sender, "You are not the owner");
    require(amount > 0, "Amount should be greater than zero");
    require(amount <= balance, "Insufficient balance");

    balance -= amount;

    (bool success, ) = payable(accOwner).call{value: amount}("");

    require(success, "Transfer failed");
}


Here, amount is the withdrawal amount in Wei.

This function:
- Checks ownership.
- Checks that the amount is greater than zero.
- Checks sufficient balance.
- Decreases the internal balance.
- Transfers Ether to the owner.
- Reverts the transaction if the transfer fails (including restoring the previous balance change).
For a production banking contract, additional security considerations and testing are necessary.
